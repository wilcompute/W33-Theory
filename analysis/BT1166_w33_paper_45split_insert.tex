\begin{theorem}[Boolean $45$ split]
\label{thm:boolean-fortyfive-split}
For the $60$-column orbit-closed Boolean/Clifford module $M$,
\[
  \operatorname{rank}(P_-M)=15.
\]
Therefore the complementary count is
\[
  60-15=45.
\]
By grade and feature type,
\[
  (16,24,16,4)=(4,6,4,1)+(12,18,12,3),
\]
and
\[
  (12,18,12,3)=3(4,6,4,1).
\]
Thus the module splits into a projected projective $15$ image and three further
copies of the same grade pattern, giving the expected $45$ side.
\end{theorem}
