\begin{abstract}
\noindent We prove that the zero-error (Shannon) channel capacity of the $W(3,3)$
collinearity graph is
\[
  \Theta\bigl(W(3,3)\bigr) = \Phi_4 = q^2+1 = 10,
  \qquad
  C_{\mathrm{ZE}} = \log_2 10 \approx 3.3219\text{ bits per channel use.}
\]
The proof combines two facts: $W(3,3)$ is vertex-transitive under $\mathrm{PSp}(4,3)$
(order $51\,840$), and for vertex-transitive graphs the zero-error capacity equals the
Lov\'asz theta function (Lov\'asz 1979). The Lov\'asz theta of $\mathrm{SRG}(40,12,2,4)$
is $\vartheta = -vs/(k-s) = 160/16 = 10$, exact.
We show that $\Phi_4 = 10$ is simultaneously the zero-error channel capacity, the
Lov\'asz theta, the string-theory critical dimension, the spread size, and the
denominator of the Kochen--Specker contextual fraction $1/10$---the machine certifies
its own contextuality by the same integer that governs its channel capacity.
All results are executable; witness: \texttt{analysis/w33\_shannon\_capacity.py}.
\end{abstract}

\section{Background: The $W(3,3)$ Channel}

The $W(3,3)$ collinearity graph $\Gamma$ is the strongly regular graph
$\mathrm{SRG}(40,12,2,4)$: 40 vertices (points of the symplectic polar space),
each adjacent to 12 others, with any two adjacent vertices sharing $\lambda=2$
common neighbours and any two non-adjacent vertices sharing $\mu=4$ common
neighbours. Its three eigenvalues are $k=12$ (multiplicity 1),
$r=2$ (multiplicity $f=24$), and $s=-4$ (multiplicity $g=15$).
The automorphism group is $\mathrm{PSp}(4,3) \cong W(E_6)^+$, of order $51\,840$.

A \textbf{channel} on vertex set $V$ is specified by the confusability graph $\Gamma$:
two symbols $x,y$ are confusable (cannot be reliably distinguished) iff $x\sim y$.
The zero-error capacity $\Theta(\Gamma)$ is the exponential growth rate of the
maximum number of mutually distinguishable codewords per channel use.

\section{Main Result}

\begin{theorem}[Zero-Error Shannon Capacity]\label{thm:main}
\[
  \Theta\bigl(W(3,3)\bigr) = 10 = \Phi_4 = q^2+1.
\]
\end{theorem}

\begin{proof}
\textbf{Step 1 (Vertex-transitivity).}
The group $\mathrm{PSp}(4,3)$, of order $51\,840$, acts faithfully on the 40 points
of $W(3,3)$ by symplectomorphisms, and this action is transitive.
Therefore $\Gamma = W(3,3)$ is vertex-transitive.

\textbf{Step 2 (Lov\'asz 1979).}
For any vertex-transitive graph $G$, $\Theta(G) = \vartheta(G)$
(the Lov\'asz theta function).

\textbf{Step 3 (Exact SRG formula).}
For $\mathrm{SRG}(v,k,\lambda,\mu)$ with negative eigenvalue $s < 0$,
\[
  \vartheta(\Gamma) = -\frac{v\,s}{k-s}.
\]
Substituting $v=40$, $k=12$, $s=-4$:
\[
  \vartheta\bigl(W(3,3)\bigr) = -\frac{40\cdot(-4)}{12-(-4)} = \frac{160}{16} = 10 = \Phi_4.
\]
Combining Steps 1--3: $\Theta(W(3,3)) = 10$.  \qed

\emph{Note on $\alpha(W(3,3))$:} By the Thas no-ovoid theorem, $W(q)$ has no ovoid
when $q$ is odd. For $W(3,3)$ the maximum partial ovoid has size $7 < 10$.
Thus the Hoffman bound $\alpha \le v|s|/(k+|s|) = 10$ is \emph{not} tight for $\alpha$,
but it \emph{is} tight for $\vartheta = \Theta$ via vertex-transitivity.
\end{proof}

\begin{corollary}[Capacity in bits]
\[
  C_{\mathrm{ZE}} = \log_2 \Theta = \log_2 10 \approx 3.3219\text{ bits/use.}
\]
\end{corollary}

\section{Capacity--Contextuality Identity}

\begin{theorem}[One Integer, Two Witnesses]\label{thm:context}
\[
  \Theta\bigl(W(3,3)\bigr) = \Phi_4 = 10, \qquad
  \mathrm{CF}\bigl(W(3,3)\bigr) = \frac{1}{\Phi_4} = \frac{1}{10}.
\]
The zero-error channel capacity and the Kochen--Specker contextual fraction
are governed by the same substrate integer $\Phi_4 = q^2+1$.
\end{theorem}

\begin{proof}
$\mathrm{CF} = (40-36)/40 = 1/10$ (max contexts satisfiable by a non-contextual
assignment is 36 of 40, proved by integer programming and Thas no-ovoid;
see witness \texttt{analysis/w33\_contextual\_fraction.py}).
$\Theta = 10$ by Theorem~\ref{thm:main}. Both equal $\Phi_4$.
\end{proof}

\begin{keybox}{The machine certifies itself}
The $W(3,3)$ substrate uses the same integer $\Phi_4 = 10$ for two independent
physical tasks: (i) the maximum zero-error information rate through the channel,
and (ii) the fraction of measurement rounds that cannot be explained by any classical
model. A single benchtop measurement of the contextual fraction $1/10$ simultaneously
certifies the channel's information-theoretic capacity and the quantum nature of
the substrate.
\end{keybox}
