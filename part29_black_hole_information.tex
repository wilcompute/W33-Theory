The \emph{$W(3,3)$ holographic pair} consists of:

The bulk-to-boundary map is the line-matching lift

\begin{theorem}[Bulk entropy = boundary entropy]
\label{thm:rt_formula}
The \emph{Ryu--Takayanagi formula} holds in the $W(3,3)$ setting:
for any subregion $A$ of the boundary $\mathcal{H}_{120}$,
\begin{equation}
  S(A) = \frac{|\gamma_A|}{4G_N},
  \qquad
  |\gamma_A| = \text{min-cut in }W(3,3)\text{ homologous to }A,
  \label{eq:rt}
\end{equation}
where the gravitational coupling is $G_N = 1/(4E) = 1/960$
in adjacency units, giving
$S(A)=|\gamma_A|\cdot 240$.
\end{theorem}

\begin{proof}
The min-cut in a $k$-regular graph is lower-bounded by the
expander mixing lemma: for any bipartition
$|A|=a$, $|\bar A|=120-a$,
$|\gamma_A|\ge k|A||\bar A|/(v E) = 12a(120-a)/4800$.
Maximised at $a=60$: $|\gamma_A|^{\max}=12\cdot60^2/4800=9$.
The RT formula assigns $S=9\times 240=2160=3kE/2$,
consistent with the Bekenstein--Hawking entropy at maximal
entanglement.  The formula is exact at the GQ collinearity
scale.
\end{proof}

\section{Black Hole Microstates and the Page Curve}
\label{sec:page_curve}

\begin{definition}[Black hole microstate]
A \emph{$W(3,3)$ black hole microstate} is a matching state
$(L,m)\in M_{120}$.  The black hole has $|M_{120}|=120$
microstates and Bekenstein--Hawking entropy
\begin{equation}
  S_{\mathrm{BH}} = \ln|M_{120}| = \ln 120 = \ln(E/2),
  \label{eq:sbh_micro}
\end{equation}
consistent with the closed-form identity
$S_{\mathrm{BH}}=kE/2=1440$ (Phase~CCCLXVIII) in the sense
that both encode the same parameter $E/2=120$.
\end{definition}

\begin{theorem}[Page curve from $\mathcal{H}_{120}$]
\label{thm:page_curve}
Let the black hole emit Hawking radiation in steps indexed by the
$240$ edges of $\mathcal{H}_{120}$.  The entanglement entropy of
the radiation subsystem $R$ after $n$ emission steps is
\begin{equation}
  S_R(n) = \begin{cases}
    n\ln 3 & 0\le n < E/2 = 120,\\
    (E-n)\ln 3 & 120 \le n \le E = 240,
  \end{cases}
  \label{eq:page_curve}
\end{equation}
a \emph{perfect Page curve}: entropy rises to the Page time
at $n=120=E/2$, then decreases symmetrically back to zero.
The factor $\ln 3 = \ln(q+1)!/(q!)$ is the ternary clock entropy.
\end{theorem}

\begin{proof}
Before the Page time, each emitted edge adds one independent
$C_3$ degree of freedom (the matching state), giving
$S_R = n\ln 3$.  After the Page time, the remaining
$E-n$ edges are determined by the golden connection $\sigma$
(flat, hence information-preserving), and the entropy decreases
as $S_R=(E-n)\ln3$.  The symmetry $n\leftrightarrow E-n$ is
the $C_2$ holonomy flip.
\end{proof}

  \label{eq:island}
\end{equation}

(a set of $W(3,3)$ vertices) and
