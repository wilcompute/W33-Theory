\section{Adjacency Smith Groups and the 2-Adic Transfer Law}
\label{sec:smith}

\subsection{The Smith Normal Form Invariant}

Let $G$ be a graph with nonsingular adjacency matrix $A$. Define its
\emph{adjacency Smith group}
\[
 K_A(G)=\operatorname{coker}(A)=\mathbb Z^n/A\mathbb Z^n.
\]
This is not the graph critical (sandpile) group, which is defined from a
reduced Laplacian. The group $K_A(G)$ is determined up to isomorphism by the
Smith normal form (SNF) of $A$.
Write $\mathrm{SNF}(A) = \mathrm{diag}(s_1, \ldots, s_n)$ with $s_i \mid s_{i+1}$.

For a strongly regular graph with parameters $(v,k,\lambda,\mu)$, write the
adjacency spectrum as $\{k^1,r^f,s^g\}$. Then
\[
 |\det(A)|=|k r^f s^g|
\]
is a spectral invariant.

\subsection{The 2-Adic Transfer Theorem}

\begin{theorem}[$p$-Adic Transfer Law]
\label{thm:transfer}
Let $G_1, G_2$ be cospectral graphs (same adjacency eigenvalue multiset).
For every prime $p$:
$$\sum_{i=1}^{n} v_p(s_i(G_1)) = \sum_{i=1}^{n} v_p(s_i(G_2)).$$
That is, the total $p$-adic valuation of the adjacency Smith group order is
a spectral invariant.
\end{theorem}

\begin{proof}
We have $\det(A) = \prod_i \lambda_i$ (product over eigenvalues, a spectral invariant)
and $\det(A) = \prod_i s_i$ (product over SNF diagonal entries).
Hence $v_p(\det A) = \sum_i v_p(\lambda_i) = \sum_i v_p(s_i)$,
and the right-hand side is the same for any cospectral pair. \qed
\end{proof}

\begin{corollary}
For any nonsingular cospectral pair $G_1,G_2$,
$|K_A(G_1)|_p=|K_A(G_2)|_p$ for all primes $p$, where $|\cdot|_p$
denotes the $p$-part of the group order. The theorem fixes the total
valuation, not the individual invariant factors.
\end{corollary}

\subsection{The Certified W33 Smith Data}

The Spence family consists of 28 non-isomorphic $\mathrm{SRG}(40,12,2,4)$, including
the symplectic polar space $W(3,3)$ and the orthogonal polar space $Q(4,3)$.
All share the adjacency spectrum $\{12^1, 2^{24}, (-4)^{15}\}$, hence by
Theorem~\ref{thm:transfer}, all have $v_2(\mathrm{Smith}) = v_2(\det A) = 56$.

\begin{theorem}[W33 adjacency Smith group]
\label{thm:ladder}
The verified Smith normal form is
\[
 \operatorname{SNF}(A_{W33})
 =\operatorname{diag}(1^{16},2^8,8^{15},24).
\]
Consequently $\operatorname{rank}_{\mathbb F_2}(A_{W33})=16$ and its
mod-$2$ nullity is $24$. Moreover
$|K_A(W33)|=3\cdot2^{56}$, as required by the spectrum.
\end{theorem}

\begin{proposition}[Open all-28 census]
\label{thm:transcend}
The complete Smith normal forms of all 28 Spence graphs have not been
computed in the committed certificate. In particular, the proposed
$\{17,8,2,1\}$ class multiplicities remain a target rather than a theorem.
For any integer matrix with Smith entries $s_i$,
\[
 \operatorname{rank}_{\mathbb F_p}(A)=\#\{i:p\nmid s_i\};
\]
counting entries divisible by $p$ gives the nullity, not the rank.
\end{proposition}

\begin{remark}
For regular graphs the ordinary Ihara zeta and the one-variable
Hashimoto determinant are determined by the adjacency spectrum, as is the
standard Bartholdi specialization. A differing adjacency SNF would therefore
refine those spectral invariants. This does not imply that it refines every
``edge zeta'': multivariate edge-zeta constructions can retain
non-spectral incidence information.
\end{remark}
