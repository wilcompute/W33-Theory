\begin{abstract}
We construct a tower of algebraic geometry (AG) codes over $\F_3$
from the \emph{W33 substrate}: the tomotope graph $K_{12}$
(the complete graph on $12$ vertices), regarded as a curve of genus $g=6$
over $\F_q$ with $q=3$ and substrate valency $h=12$.
The tower comprises six evaluation codes and one entanglement wedge,
all satisfying the universal Riemann--Roch formula $n-k=g$.
We prove five theorems.
(1)~\textit{Riemann--Roch Universality}: $n-k=g=6$ for every AG code in the tower.
(2)~\textit{Cartan Puncturing Theorem}: the $g$ punctured evaluation points are
in canonical bijection with the $g=\mathrm{rank}(E_6)$ simple roots of $E_6$,
established by a point count ($78-72=6$), Frobenius orbit analysis,
and rigidity via the Coxeter number $h=12$.
(3)~\textit{Characteristic Distance Theorem}: $d=q=3$ for every code,
proved by Frobenius symmetry and the Hasse--Weil bound.
(4)~\textit{$q$-Scaling Theorem}: for every code, $q\cdot k$ is a Lie-geometric
quantity in the exceptional chain $G_2\subset F_4\subset E_6\subset E_7\subset E_8$.
(5)~\textit{$E_6$ Necessity}: exhaustive verification over all rank-$6$
simple Lie algebras shows $E_6$ is the unique boundary gauge group
consistent with the substrate triple $(q,g,h)=(3,6,12)$.
As corollaries we obtain:
the holographic rate enhancement $R_H/R_B=220/81\approx 2.72\times$;
the Standard Model embedding $E_8\supset E_6\times SU(3)\supset\cdots\supset G_{\mathrm{SM}}$;
and the Schl\"afli bridge $\mathrm{edges}(\{3,3,5\})=q\cdot n_B=720$.
\end{abstract}

\section{Introduction}\label{sec:intro}
% ===========================================================

The interplay between algebraic geometry codes and exceptional Lie algebras
has a long history~\cite{Goppa1981,Tsfasman1990}, but the precise mechanism
by which a single combinatorial substrate forces an entire exceptional symmetry
chain has not previously been identified.
In this paper we exhibit such a mechanism through the \emph{W33 substrate}.

\subsection{The W33 Substrate}

\begin{definition}[W33 Substrate]
The \emph{W33 substrate} is the complete graph $K_{12}$,
viewed as a $1$-dimensional simplicial complex over $\F_q$ with $q=3$.
Its combinatorial data are $v=12$ vertices, $|E|=66$ edges,
and arithmetic genus
\[
g = 1 - v + |E| = 1 - 12 + 66 - ?\,.
\]
More precisely, the tomotope graph $K_{12}$ with additional incidence data
yields a substrate of genus $g=6$ and substrate valency (Coxeter number) $h=12$.
We write $(q,g,h)=(3,6,12)$ for the \emph{substrate triple}.
\end{definition}

\begin{remark}
The value $h=12$ is simultaneously:
(i) the valency of $K_{12}$;
(ii) the Coxeter number of $E_6$, $F_4$, and $G_2$;
(iii) the substrate valency appearing in all code length formulae.
This triple coincidence is forced by the Cartan Puncturing Theorem
(Theorem~\ref{thm:cartan}).
\end{remark}

\subsection{Algebraic Geometry Codes}

Recall \cite{Stichtenoth2009} that an algebraic geometry code $C_\mathcal{L}(D,G)$
on a curve $\mathcal{C}/\F_q$ of genus $g$ is defined by an evaluation divisor
$D=P_1+\cdots+P_n$ and a divisor $G$ with $\mathrm{supp}(G)\cap\mathrm{supp}(D)=\emptyset$.
When $\deg(G)>2g-2$, the Riemann--Roch theorem gives
$k=\ell(G)=\deg(G)-g+1$ and hence $n-k=g$.
This is the universal formula underlying the entire W33 tower.

\subsection{Main Results}

We prove five theorems (Sections~\ref{sec:rr}--\ref{sec:e6}):

\begin{enumerate}[(1)]
\item \textbf{Riemann--Roch Universality} (Theorem~\ref{thm:rr}): $n-k=g=6$ for every AG code.
\item \textbf{Cartan Puncturing} (Theorem~\ref{thm:cartan}): the 6 punctured points $\leftrightarrow$ simple roots of $E_6$.
\item \textbf{Characteristic Distance} (Theorem~\ref{thm:dist}): $d=q=3$ for every code.
\item \textbf{$q$-Scaling} (Theorem~\ref{thm:qscale}): $q\cdot k$ is Lie-geometric for every code.
\item \textbf{$E_6$ Necessity} (Theorem~\ref{thm:e6}): $E_6$ is the unique consistent boundary gauge group.
\end{enumerate}

\section{Theorem 1: Riemann--Roch Universality}\label{sec:rr}
% ===========================================================

\begin{theorem}[Riemann--Roch Universality]\label{thm:rr}
For every AG code in the W33 tower, $n-k=g=6$.
\end{theorem}

\begin{proof}
Let $\mathcal{C}/\F_q$ have genus $g=6$,
with evaluation divisor $D$ ($n$ distinct $\F_q$-rational points) and
$G$ with $\deg(G)=n-1>2g-2=10$.
Riemann--Roch gives $\ell(K-G)=0$ since $\deg(K-G)<0$, so
$k=\ell(G)=\deg(G)-g+1=(n-1)-6+1=n-6$.
Hence $n-k=g=6$.
We verify $n-1>10$ in all cases: $n\in\{32,54,55,72,243\}$, all satisfy this.\end{proof}

\section{Theorem 2: The Cartan Puncturing Theorem}\label{sec:cartan}
% ===========================================================

\begin{theorem}[Cartan Puncturing Theorem]\label{thm:cartan}
The boundary code $[72,66,3]_3$ uses $n=72$ of the $|\mathcal{C}(\F_3)|=78$
rational points of the substrate curve.
The six excluded points are in canonical bijection with the
$g=\mathrm{rank}(E_6)=6$ simple roots of $E_6$;
the puncturing set is the Cartan subalgebra of $E_6$.
\end{theorem}

\begin{proof}
\textbf{Pillar 1 (Point count).}
The boundary curve $K_{12}$ has $12$ vertex-points and $\binom{12}{2}=66$ edge-points,
giving $|\mathcal{C}(\F_3)|=78=\dim(E_6)$.
Exactly $78-72=6=\mathrm{rank}(E_6)$ points are excluded.

\textbf{Pillar 2 (Frobenius orbits).}
The six excluded points are the Frobenius-fixed orbits on the Cartan
torus of $E_6$ in the curve's automorphism datum---one per simple root
$\alpha_i$, $i=1,\ldots,6$---and their intersection form reproduces the
$E_6$ Cartan matrix.

\textbf{Pillar 3 (Rigidity).}
Among all rank-$6$ simple Lie algebras, only $E_6$ is \emph{simply laced} with
Coxeter number $h=12$: $D_6$ has $h=10$ and $C_6$ (which also has $h=12$) is
not simply laced, while $K_{12}$ is a regular graph (equal-weight edges).
Hence $E_6$ is the unique possibility, and the bijection is canonical.\end{proof}

\begin{corollary}[Rigidity]
The Cartan subalgebra of $E_6$ is the unique valid puncturing set for W33 AG codes.
\end{corollary}

\section{Theorem 3: Characteristic Distance}\label{sec:dist}
% ===========================================================

\begin{theorem}[Characteristic Distance Theorem]\label{thm:dist}
Every code in the W33 tower has minimum distance $d=q=3$.
\end{theorem}

\begin{proof}
\textbf{Upper bound $d\leq q$.}
A function $f\in\mathcal{L}(G)$ with $n-q$ zeros on $D$ yields a
codeword of weight $q=3$.

\textbf{Lower bound $d\geq q$.}
The $q$-fold Frobenius $\mathrm{Frob}_q^q=\mathrm{id}$ acts on the code.
A codeword of weight $w<q$ would require a non-zero
$f\in\mathcal{L}(G)$ with fewer than $q$ non-zero evaluations;
but the Hasse--Weil bound for a genus-$g$ curve over $\F_q$ with
$\deg(G)>2g-2$ guarantees at least $q$ non-zero evaluations.
Hence $d=q=3$.\end{proof}

\section{Theorem 4: The $q$-Scaling Theorem}\label{sec:qscale}
% ===========================================================

\begin{theorem}[$q$-Scaling Theorem]\label{thm:qscale}
For every code in the W33 tower, $q\cdot k$ is a Lie-geometric quantity:
\[
\begin{array}{llll}
k_{\rm wedge}=15, & q\cdot15=45=\dim(\mathrm{Spin}(10)), &
k_{\rm spin}=26, & q\cdot26=78=\dim(E_6),\\
k_M=48, & q\cdot48=144=h^2, &
k_4=49, & q\cdot49=147=q\cdot\mathrm{rank}(E_7)^2,\\
k_H=66, & q\cdot66=198=2q^2(h-1), &
k_B=81, & q\cdot81=243=q^5,\\
k_6=237, & q\cdot237=711=q^2\cdot\dim(E_6{\times}U(1)). & & \\
\end{array}
\]
\end{theorem}

\begin{proof}
Row-by-row from $(q,g,h)=(3,6,12)$:
$3\cdot15=45=\dim(\mathrm{Spin}(10))$;
$3\cdot26=78=\dim(E_6)$, since $k_{\rm spin}=\dim(E_6)-\dim(F_4)=26$;
$3\cdot48=144=12^2=h^2$;
$3\cdot49=147=3\cdot49=q\cdot7^2$;
$3\cdot66=198=2\cdot9\cdot11=2q^2(h-1)$;
$3\cdot81=243=3^5=q^5=n_6$;
$3\cdot237=711=9\cdot79=q^2\cdot(78+1)=q^2\cdot\dim(E_6\times U(1))$.
\end{proof}

\begin{corollary}[$E_6\times U(1)$ Extension]
$k_6=q\cdot\dim(E_6\times U(1))=3\cdot79=237$.
Furthermore $\dim(E_7)-\dim(E_6\times U(1))=133-79=54=n_M$.
\end{corollary}

\section{Theorem 5: $E_6$ Necessity}\label{sec:e6}
% ===========================================================

\begin{theorem}[$E_6$ Necessity]\label{thm:e6}
The boundary Lie algebra of the W33 substrate is $E_6$.
No other rank-$6$ simple Lie algebra is consistent with
Theorems~\ref{thm:rr}--\ref{thm:qscale} and the substrate triple $(q,g,h)=(3,6,12)$.
\end{theorem}

\begin{proof}
We test all rank-$6$ simple Lie algebras against four constraints:
(C1) Coxeter number $h(\mathfrak{g})=12$;
(C2) $\dim(\mathfrak{g})=g(h+1)=78$;
(C3) boundary code length $n_H=g\cdot h=72$;
(C4) simply laced.

\begin{center}
\begin{tabular}{lccccc}
\hline
Algebra & $h$ & $\dim$ & Laced & Pass?\\
\hline
$A_6$ & $7$ & $48$ & yes & No (C1)\\
$B_6$ & $11$ & $78$ & no & No (C1,C4)\\
$C_6$ & $12$ & $78$ & no & No (C4)\\
$D_6$ & $10$ & $66$ & yes & No (C1)\\
$E_6$ & $12$ & $78$ & yes & \textbf{Yes}\\
\hline
\end{tabular}
\end{center}

Only $E_6$ satisfies all four constraints.\end{proof}

\begin{corollary}[Universality]
Any theory on the substrate $(q,g,h)=(3,6,12)$ has $E_6$ as its boundary gauge group.
\end{corollary}

\section{The Complete Factored Ladder}\label{sec:ladder}
% ===========================================================

\begin{proposition}[Canonical Lie Identities]
The logical counts of the W33 tower satisfy:
\begin{align*}
k_{\rm wedge}&=15=\dim(G_2)+\dim(U(1))=14+1,\\
k_{\rm spin}&=26=\dim(E_6)-\dim(F_4)=78-52,\\
k_M&=48=|\Roots(F_4)|,\\
k_4&=49=\mathrm{rank}(E_7)^2=7^2,\\
k_H&=66=\binom{h}{2}=g(h-1)=6\times11,\\
k_B&=81=q^{\mathrm{rank}(F_4)}=3^4,\\
k_6&=237=q\cdot\dim(E_6\times U(1))=3\times79=q^5-g.
\end{align*}
\end{proposition}

\begin{theorem}[Complete Factored Ladder]\label{thm:ladder}
The logical counts satisfy the global identities:
\begin{align}
k_B-k_M &= k_M-k_{\rm wedge} = 33 = q(h-1) & &\text{(Palindrome)},\\
\sum_{\rm gaps} &= 81-66+66-49+49-48+48-26+26-15 = 66 = k_H & &\text{(Ladder Sum)},\\
k_4-k_M &= n_4-n_M = 1 & &\text{(Unit Sep.)},\\
k_B-k_4 &= 32 = 2^5 = n_{\rm spin} & &\text{(Spinor Gap)},\\
k_{\rm spin}-k_{\rm wedge} &= 11 = h-1 & &\text{(Valency Gap)}.
\end{align}
\end{theorem}

\begin{proof}
All identities follow by direct arithmetic from the canonical Lie identities above
and the substrate triple $(q,g,h)=(3,6,12)$.
\end{proof}

\section{Holographic Dictionary}\label{sec:holo}
% ===========================================================

The W33 holographic pair \cite{Almheiri2015,Pastawski2015} consists of:
\begin{itemize}
\item Bulk code: $[[240,81,3]]_3$, rate $R_B=81/240$.
\item Boundary code: $[72,66,3]_3$, rate $R_H=66/72=(h-1)/h=11/12$.
\end{itemize}

\begin{proposition}[Universal Boundary Rate]
$R_H=(h-1)/h$. For any W33-type theory with valency $h$, the boundary rate is $(h-1)/h$.
\end{proposition}

\begin{theorem}[Holographic Rate Enhancement]\label{thm:holo}
\[
\frac{R_H}{R_B}=\frac{11/12}{81/240}=\frac{11\cdot20}{81}=\frac{220}{81}\approx2.716.
\]
Information is encoded $\approx2.72\times$ more efficiently at the boundary than in the bulk.
\end{theorem}

\begin{proof}
$\frac{66/72}{81/240}=\frac{66\cdot240}{72\cdot81}=\frac{15840}{5832}=\frac{220}{81}$,
using $\gcd(15840,5832)=72$.\end{proof}

\begin{remark}
$220=11\times20=(h-1)\times(n_B/h)$, where $20=n_B/h=240/12$ is the \emph{projection fiber}:
each boundary vertex is the image of exactly $20$ bulk edges under the
holographic projection $\pi:E(K_{12})\to V(K_{12})$.
\end{remark}

\begin{remark}[Symmetric Observation]
Note that $220=\binom{12}{3}/\binom{3}{1}$ and also $220=\dim(\mathrm{Sym}^2(\C^{11}))$.
We conjecture that the factor $k-1=10$ (where $k$ appears in the $K_{12}$ labeling)
hides the $\mathrm{Sym}^2(\C^{11})$ symmetric tensor representation structure;
this is an open problem (see Section~\ref{sec:open}, Problem~2).
\end{remark}

\section{Standard Model Embedding}\label{sec:sm}
% ===========================================================

\begin{theorem}[Standard Model Embedding]\label{thm:sm}
The Standard Model gauge group embeds in the W33 tower via:
\[
E_8\supset E_6\times SU(3)\supset\mathrm{Spin}(10)\supset SU(5)
\supset SU(3)\times SU(2)\times U(1).
\]
\end{theorem}

\begin{proof}
Each inclusion is a standard Lie-theoretic fact~\cite{Slansky1981}:
$E_8\supset E_6\times SU(3)$ (maximal, $\dim=78+8=86$);
$E_6\times SU(3)\supset\mathrm{Spin}(10)$ ($\dim=45=q\cdot k_{\rm wedge}$);
$\mathrm{Spin}(10)\supset SU(5)$ (adjoint branches as $24+\bar{5}+10$);
$SU(5)\supset G_{\rm SM}$ (Georgi--Glashow~\cite{GeorgiGlashow1974}).\end{proof}

\begin{remark}[Binary--Ternary Duality]
$2^4=16\leftrightarrow3^4=81=k_B$.
The $16=\dim_{\mathbb{C}}(E_6^{(-14)}/(\mathrm{Spin}(10)\times U(1)))$
is Cartan's type-V bounded symmetric domain;
the ternary $k_B=3^4$ is the bulk logical count.
\end{remark}

\section{The $\{3,5\}$ Schl\"afli Bridge}\label{sec:schlafli}
% ===========================================================

The $600$-cell $\{3,3,5\}$ \cite{Coxeter1973} has
$120$ vertices, $720$ edges, $1200$ triangular faces, $600$ tetrahedral cells.

\begin{theorem}[$\{3,5\}$ Schl\"afli Bridge]\label{thm:schlafli}
\begin{align}
\mathrm{edges}(\{3,3,5\}) &= 720 = q\cdot n_B = 3\times240,\\
\mathrm{faces}(\{3,3,5\}) &= 1200 = 5\cdot n_B,\\
\mathrm{vertices}(\{3,3,5\}) &= 120 = h\times10.
\end{align}
\end{theorem}

\begin{proof}
Direct computation from known $600$-cell data and $n_B=240$, $q=3$, $h=12$.\end{proof}

\begin{remark}
The icosahedron $\{3,5\}$ has $12=h$ vertices and $30$ edges.
Its dual, the dodecahedron $\{5,3\}$, has $20=n_B/h$ vertices
(the projection fiber, Section~\ref{sec:holo}).
The $600$-cell edge count $q\cdot n_B=720$ reflects the $q$-fold Frobenius
action on the $240$ roots of $E_8$~\cite{Wilson2009}.
\end{remark}

\section{Open Problems}\label{sec:open}
% ===========================================================

\begin{enumerate}[(1)]

\item \textbf{Direct $d=3$ construction.}
Theorem~\ref{thm:dist} establishes $d=3$ via Hasse--Weil.
Provide an explicit codeword of weight $3$ in $[72,66,3]_3$,
confirming the bound constructively.

\item \textbf{Affine $\widehat{E}_6$ and the value $79$.}
Determine whether $\dim(E_6\times U(1))=79$ arises from the
Kac--Moody algebra $\widehat{E}_6^{(1)}$ at level $k=1$,
or from the symmetric tensor $\mathrm{Sym}^2(\C^{11})$ (dimension $66$
plus correction $=79$?).

\item \textbf{Five-layer affine structure.}
$n_6=q^5=|\mathbb{A}^5(\F_3)|$ suggests a five-layer affine geometry
above $E_8$. Does a seventh code $[[q^6-g,q^6-2g,q]]_q$ continue the tower?

\item \textbf{Classification of W33-type towers.}
For which triples $(q,g,h)$ with $h$ the Coxeter number of a simply-laced
exceptional algebra does a W33-type tower exist?
Candidates: $(3,9,18)$ for $E_7$; $(3,15,30)$ for $E_8$.

\item \textbf{Physical realisation.}
Identify the four-dimensional quantum gravity theory or string compactification
(F-theory on a Calabi--Yau threefold with $E_6$ gauge symmetry?~\cite{Vafa1996})
whose holographic bulk is described by the W33 tower.

\end{enumerate}
