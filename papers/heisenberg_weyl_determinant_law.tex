\title{A determinant congruence for Heisenberg--Weyl blocks}
\author{W.~Dahn}
\date{July 2026}

\begin{document}
\maketitle

\begin{abstract}
\noindent Let $q$ be an odd prime, $H$ the Heisenberg group of order $q^3$, and
$\psi_t$ a nontrivial character of its centre. An \emph{inverse-closed
section} $c$ assigns to each nonzero $v\in\FF_q^2$ a central coordinate
$c(v)\in\FF_q$ with $c(-v)=-c(v)$; the associated \emph{Weyl block}
$B_t(c)$ is the image, in the $q$-dimensional irreducible representation
attached to $\psi_t$, of the sum of the corresponding group elements. We prove
that the first-order variation of $\det B_t(c)$ about the zero section is
divisible by $\lambda^{\vlam(q)+2}$, where $\lambda=1-\zeta_p$, and that the
exponent splits canonically as
\[
\vlam(q) \;+\; 2,\qquad
\underbrace{\vlam(q)}_{\text{ramification of }q}
\;+\;\underbrace{2}_{\text{inverse closure}} .
\]
The proof rests on a closed formula
$\tr\bigl(\adj(F)\,D\bigr)=\det(F)\,q\,S/(q^2-1)$ in which $F$ is the block of
the zero section and $S=\sum_{v}(\zeta_q^{-tc(v)}-1)$; the block $F$ is shown to
satisfy $F^2+2F-(q^2-1)I=0$ by a symplectic character sum. We then prove the
congruence itself, in the form
\[
\det B_t(c)\;\equiv\;\det F \pmod{\lambda^{\,\vlam(q)+4}},
\]
valid for every odd prime \emph{power} $q=p^{f}$ and every section, modulo one
explicitly identified lemma on the top exterior power. Here
$\vlam(q)=f(p-1)$, so the exponent is $q+3$ when $q$ is prime and $8$ when
$q=9$. The constant $4$ splits as $2+2$: two units of vanishing from inverse
closure at first order, and two more from a cancellation between the first two
orders whose mechanism is that a \emph{symmetric} coefficient is paired against
the \emph{antisymmetric} symplectic form. Neither summand depends on the
factorization of $q$; only the ramification does. Several steps of the
argument would fail on valuation counting alone, and each is rescued by
inverse closure.
\end{abstract}

valuation, ramification, determinant congruence, Gauss sums.

\section{Introduction}

Fix an odd prime $q$ and let
\[
H=\bigl\{(a,b,c):a,b,c\in\FF_q\bigr\},\qquad
(a,b,c)(a',b',c')=(a+a',\,b+b',\,c+c'-ab'+a'b),
\]
the extraspecial group of order $q^3$ and exponent $q$, with centre
$Z=\{(0,0,c)\}$. Each nontrivial character $\psi_t(0,0,c)=\zeta^{tc}$
($\zeta=e^{2\pi i/q}$, $t\not\equiv0$) determines a $q$-dimensional irreducible
representation $\rho_t$ of $H$, realized on $\mathbb C^{\FF_q}$ by
\begin{equation}\label{eq:rho}
\rho_t(a,b,c)\,e_x \;=\; \zeta^{\,t(c+2xb+ab)}\,e_{x+a}.
\end{equation}

Call a function $c:\FF_q^2\setminus\{0\}\to\FF_q$ an \emph{inverse-closed
section} if $c(-v)=-c(v)$; equivalently, the set
$S(c)=\{(v,c(v)):v\neq0\}\subset H$ is closed under inversion, since
$(a,b,z)^{-1}=(-a,-b,-z)$. Such sections index the ways of lifting the
punctured plane $\FF_q^2\setminus\{0\}$ into $H$ compatibly with inversion,
and the object of interest is the \emph{Weyl block}
\[
B_t(c)\;=\;\sum_{v\in\FF_q^2\setminus\{0\}}\rho_t\bigl(v,c(v)\bigr)
\;\in\; M_q\bigl(\ZZ[\zeta]\bigr),
\]
a Hermitian matrix over the cyclotomic ring $\ZZ[\zeta]$. Write $F=B_t(0)$ for
the block of the zero (\emph{flat}) section.

Sections arise naturally: $S(c)$ together with $Z\setminus\{1\}$ is the Cayley
set of a strongly regular graph on $H$ with parameters
$\bigl(q^3,(q-1)(q+2),q-2,q+2\bigr)$ exactly when $c$ is flat up to the obvious
symmetries, and the family of these partial difference sets in $H$ is
classical \cite{Ma,PDSS}. The determinants $\det B_t(c)$, as $c$ ranges over
all sections, are therefore a natural arithmetic invariant of that family. Our
results describe how little they vary.

Throughout, $\lambda=1-\zeta$ is the unique prime of $\ZZ[\zeta]$ above $q$, so
that $(q)=(\lambda)^{q-1}$ and
\begin{equation}\label{eq:vq}
\vlam(q)=q-1,
\end{equation}
and $\vlam$ denotes the $\lambda$-adic valuation.

\begin{theorem}[first-order law]\label{thm:main}
Let $q$ be an odd prime, $t\not\equiv0$, and $c$ an inverse-closed section. Put
$D=B_t(c)-F$ and $T_1=\tr\bigl(\adj(F)\,D\bigr)$, the first-order term of
$\det(F+D)$. Then
\[
T_1\;=\;\frac{\det(F)\;q\;S}{q^2-1},
\qquad
S=\sum_{v\in\FF_q^2\setminus\{0\}}\bigl(\zeta^{-t c(v)}-1\bigr),
\]
and consequently
\[
\vlam(T_1)\;=\;(q-1)+\vlam(S)\;\ge\;(q-1)+2\;=\;q+1 .
\]
\end{theorem}

The two summands are structurally distinct: $q-1=\vlam(q)$ is the ramification
index \eqref{eq:vq}, while the $2$ comes from inverse closure through the
factorization $w+w^{-1}-2=-(1-w)(1-w^{-1})$ applied to $w=\zeta^{tc(v)}$. Only
the first depends on $q$ being prime; writing the bound as $\vlam(q)+2$ makes
the statement correct for every prime power (Section~\ref{sec:pf}).

Section~\ref{sec:sharp} proves the congruence at exponent $\vlam(q)+2$
(Theorem~\ref{thm:cong}) and then lifts it to the sharp $\vlam(q)+4$
(Theorems~\ref{thm:cancel} and \ref{thm:sharp}), modulo a single explicitly
identified bound on the top exterior power. Section~\ref{sec:pf} explains why
no separate prime-power statement is needed.

\section{The flat block}

\begin{lemma}\label{lem:trace}
For every noncentral $g\in H$ and every $t\not\equiv0$, $\tr\rho_t(g)=0$.
Consequently $\tr F=0$ and $\tr B_t(c)=0$ for every section $c$.
\end{lemma}

\begin{proof}
Write $g=(a,b,c)$. If $a\neq0$ then by \eqref{eq:rho} the matrix $\rho_t(g)$
moves every basis vector $e_x$ to $e_{x+a}\neq e_x$, so its diagonal vanishes.
If $a=0$ and $b\neq0$ then $\rho_t(0,b,c)$ is diagonal with entries
$\zeta^{t(c+2xb)}$, and
$\tr\rho_t(0,b,c)=\zeta^{tc}\sum_{x\in\FF_q}\zeta^{2tbx}=0$
because $2tb\neq0$ in $\FF_q$ ($q$ is odd). Every element of $S(c)$ is
noncentral, whence the last claim.
\end{proof}

Let $\omega(v,w)=ab'-a'b$ denote the symplectic form on $\FF_q^2$, where
$v=(a,b)$ and $w=(a',b')$, so that $(v,0)(w,0)=(v+w,\,-\omega(v,w))$.

\begin{lemma}[flat block]\label{lem:flat}
$F^2+2F-(q^2-1)I=0$. Hence $F$ has spectrum
$\{(q-1)^{(q+1)/2},\,(-(q+1))^{(q-1)/2}\}$,
\[
\det F=(q-1)^{(q+1)/2}\bigl(-(q+1)\bigr)^{(q-1)/2},
\qquad
F^{-1}=\frac{F+2I}{q^2-1},
\]
and $\det F$ is a unit at $\lambda$.
\end{lemma}

\begin{proof}
Expanding and using $\rho_t(u,z)=\zeta^{tz}\rho_t(u,0)$,
\[
F^2=\sum_{v,w\neq0}\rho_t\bigl((v,0)(w,0)\bigr)
    =\sum_{v,w\neq0}\zeta^{-t\,\omega(v,w)}\,\rho_t(v+w,0).
\]
Split according to $u=v+w$. The terms with $u=0$ have $w=-v$ and
$\omega(v,-v)=0$, contributing $(q^2-1)I$. For fixed $u\neq0$ the constraint is
$v\notin\{0,u\}$, and $\omega(v,u-v)=\omega(v,u)$, so those terms contribute
\[
\Bigl(\sum_{v\neq0,u}\zeta^{-t\,\omega(v,u)}\Bigr)\rho_t(u,0).
\]
Since $u\neq0$, the map $v\mapsto\omega(v,u)$ is a nonzero linear functional on
$\FF_q^2$, so $\sum_{v\in\FF_q^2}\zeta^{-t\omega(v,u)}=0$; removing the two
excluded values $v=0$ and $v=u$, both of which give $\omega=0$, leaves
$-2$. Summing over $u\neq0$ gives $-2F$, so $F^2=(q^2-1)I-2F$.

Thus the minimal polynomial of $F$ divides
$x^2+2x-(q^2-1)=(x-(q-1))(x+(q+1))$, so the eigenvalues lie in
$\{q-1,-(q+1)\}$. If they occur with multiplicities $\alpha,\beta$ then
$\alpha+\beta=q$ and, by Lemma~\ref{lem:trace},
$\alpha(q-1)-\beta(q+1)=0$; solving gives $\alpha=(q+1)/2$,
$\beta=(q-1)/2$. The remaining assertions follow, the last because $q\pm1$ are
prime to $q$.
\end{proof}

\begin{remark}
Lemma~\ref{lem:flat} also reproves the two ``trace laws''
$\tr F=0$ and $\tr F^2=q(q^2-1)$; the latter is
$(q-1)^2\frac{q+1}{2}+(q+1)^2\frac{q-1}{2}=q(q^2-1)$. These hold for every
section, not only the flat one, and are used below.
\end{remark}

\section{The first-order law}

\begin{lemma}[pairing]\label{lem:pair}
Let $R$ be a commutative ring and $w\in R$ a unit with inverse $w^{-1}$. Then
\[
w+w^{-1}-2=-(1-w)(1-w^{-1}).
\]
In particular, if $\mu\mid(1-w)$ and $\mu\mid(1-w^{-1})$ then
$\mu^2\mid\bigl(w+w^{-1}-2\bigr)$.
\end{lemma}

\begin{proof}
$(1-w)(1-w^{-1})=1-w-w^{-1}+ww^{-1}=2-w-w^{-1}$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Since $\det F$ is a $\lambda$-unit, $F$ is invertible and
$T_1=\tr(\adj(F)D)=\det(F)\tr(F^{-1}D)$. By Lemma~\ref{lem:flat},
$F^{-1}=(F+2I)/(q^2-1)$, and $\tr D=\tr B_t(c)-\tr F=0$ by
Lemma~\ref{lem:trace}, so
\[
T_1=\frac{\det(F)}{q^2-1}\Bigl[\tr(FD)+2\tr D\Bigr]
   =\frac{\det(F)\,\tr(FD)}{q^2-1}.
\]
Now $\tr(FD)=\tr\bigl(FB_t(c)\bigr)-\tr(F^2)=\tr\bigl(FB_t(c)\bigr)-q(q^2-1)$.
For the first term,
\[
\tr\bigl(FB_t(c)\bigr)
=\sum_{v,w\neq0}\tr\rho_t\bigl((v,0)(w,c(w))\bigr),
\qquad
(v,0)\bigl(w,c(w)\bigr)=\bigl(v+w,\;c(w)-\omega(v,w)\bigr).
\]
By Lemma~\ref{lem:trace} only central products contribute, i.e.\ only $w=-v$;
then $\omega(v,-v)=0$ and the product is $\bigl(0,0,c(-v)\bigr)=(0,0,-c(v))$,
whose trace is $q\,\zeta^{-tc(v)}$. Hence
$\tr(FB_t(c))=q\sum_{v\neq0}\zeta^{-tc(v)}$, and since there are $q^2-1$
nonzero $v$,
\[
\tr(FD)=q\sum_{v\neq0}\bigl(\zeta^{-tc(v)}-1\bigr)=q\,S,
\]
which is the displayed formula.

For the valuation, $\vlam(\det F)=\vlam(q^2-1)=0$ and $\vlam(q)=q-1$ by
\eqref{eq:vq}, so $\vlam(T_1)=(q-1)+\vlam(S)$. Group the sum defining $S$ into
the $(q^2-1)/2$ pairs $\{v,-v\}$. Using $c(-v)=-c(v)$ and
Lemma~\ref{lem:pair} with $w=\zeta^{tc(v)}$,
\[
\bigl(\zeta^{-tc(v)}-1\bigr)+\bigl(\zeta^{tc(v)}-1\bigr)
=-\bigl(1-\zeta^{tc(v)}\bigr)\bigl(1-\zeta^{-tc(v)}\bigr).
\]
Each factor is $0$ if $c(v)=0$ and has valuation $1$ otherwise, so every pair
contributes valuation $2$ or $\infty$; therefore $\vlam(S)\ge2$ and
$\vlam(T_1)\ge q+1$.
\end{proof}

\section{The congruence}\label{sec:sharp}

Write $G=F^{-1}D$, $p_m=\tr(G^m)$, and $e_k=e_k(G)$, so that
$\det B_t(c)=\det(F)\bigl(1+\sum_{k\ge1}e_k\bigr)$. Note $G$ is
$\lambda$-integral, since $F^{-1}=(F+2I)/(q^2-1)$ and $q^2-1$ is prime to
$\lambda$; hence so is every $e_k$.

\begin{lemma}[power sums]\label{lem:power}
$\vlam(p_m)\ge q+1$ for every $m\ge1$.
\end{lemma}

\begin{proof}
Put $H=(F+2I)D$, so $p_m=\tr(H^m)/(q^2-1)^m$ and
$\vlam(p_m)=\vlam(\tr H^m)$. Both $F$ and $D$ are $\ZZ[\zeta]$-linear
combinations of the matrices $\rho_t(g)$, and $D=\sum_{v\neq0}d_v\rho_t(v,0)$
with $d_v=\zeta^{tc(v)}-1$, each of valuation $\ge1$. Expanding $\tr(H^m)$
therefore produces monomials
$d_{v_1}\cdots d_{v_m}\cdot\tr\rho_t(g)$ for single group elements $g$, with
exactly $m$ factors $d$. By Lemma~\ref{lem:trace} such a trace vanishes unless
$g$ is central, in which case it equals $q$ times a root of unity. Every
surviving monomial thus carries one factor $q$ and $m$ factors $d$, so
$\vlam(\tr H^m)\ge(q-1)+m$, which is $\ge q+1$ for $m\ge2$.

For $m=1$ this count gives only $q$. But by the computation in the proof of
Theorem~\ref{thm:main}, $\tr H=\tr(FD)+2\tr D=qS$, and inverse closure gives
$\vlam(S)\ge2$; hence $\vlam(p_1)=(q-1)+\vlam(S)\ge q+1$ as well.
\end{proof}

\begin{lemma}[top exterior power]\label{lem:top}
$\vlam(e_q)\ge q+1$.
\end{lemma}

\begin{proof}
$e_q=\det G=\det D/\det F$ and $\vlam(\det F)=0$, so it suffices that
$\vlam(\det D)\ge q+1$. Every entry of $D$ is a $\ZZ[\zeta]$-combination of the
$d_v$, so $D=\lambda D'$ with $D'$ $\lambda$-integral and
$\det D=\lambda^{q}\det D'$; we must show $\det D'\equiv0\pmod\lambda$.

Reduce modulo $\lambda$, identifying $\ZZ[\zeta]/\lambda=\FF_q$ via
$\zeta\mapsto1$. Then $(\zeta^{k}-1)/(1-\zeta)\equiv-k$ and
$\rho_t(v,0)\equiv P_{a_v}$, where $P_a$ is the permutation matrix of
$x\mapsto x+a$; grouping the $q^2-1$ vectors by their first coordinate,
\[
D'\;\equiv\;\sum_{a\in\FF_q}\kappa_a P_a \pmod\lambda,
\qquad
\kappa_a=-t\sum_{b}c(a,b).
\]
The $P_a$ span the group algebra
$\FF_q[\ZZ/q]=\FF_q[y]/(y^{q}-1)=\FF_q[y]/\bigl((y-1)^{q}\bigr)$, in which the
determinant of multiplication by $f(y)$ is $f(1)^{q}$. Hence
$\det D'\equiv\bigl(\sum_a\kappa_a\bigr)^{q}
=\bigl(-t\sum_{v\neq0}c(v)\bigr)^{q}$, and inverse closure makes
$\sum_{v\neq0}c(v)=0$ because the terms cancel in pairs $\{v,-v\}$. So
$\det D'\equiv0$.
\end{proof}

\begin{theorem}[the congruence]\label{thm:cong}
For every odd prime power $q$, every $t\not\equiv0$, and every inverse-closed
section $c$,
\[
\det B_t(c)\;\equiv\;\det F\pmod{\lambda^{\,\vlam(q)+2}} .
\]
\end{theorem}

\begin{proof}
By Newton's identities, $k\,e_k=\sum_{i=1}^{k}(-1)^{i-1}e_{k-i}p_i$. For
$1\le k\le q-1$ the integer $k$ is a unit at $\lambda$, so induction on $k$
together with Lemma~\ref{lem:power} and the $\lambda$-integrality of the
$e_j$ gives $\vlam(e_k)\ge q+1$. For $k=q$ the same recursion would lose
$\vlam(q)=q-1$, and Lemma~\ref{lem:top} supplies that case directly. Hence
$\vlam(e_k)\ge q+1$ for all $k\ge1$ and
$\det B_t(c)-\det F=\det(F)\sum_{k\ge1}e_k$ has valuation $\ge q+1$.
\end{proof}

\begin{remark}
Both places where valuation counting alone is insufficient --- the first power
sum in Lemma~\ref{lem:power} and the top exterior power in
Lemma~\ref{lem:top} --- are rescued by inverse closure, in the first case
through $\vlam(S)\ge2$ and in the second through $\sum_{v}c(v)=0$. Without
that hypothesis the argument gives only $\lambda^{q}$.
\end{remark}

\subsection{The sharp exponent}

The exponent can be improved by exactly two, and the improvement is a single
cancellation which we now prove.

\begin{lemma}[parity]\label{lem:parity}
For odd $m$, $\vlam(p_m)\ge\vlam(q)+m+1$. In particular
$\vlam(p_3)\ge\vlam(q)+4$.
\end{lemma}

\begin{proof}
By the proof of Lemma~\ref{lem:power} the coefficient of the leading power of
$\lambda$ in $\tr(H^m)$ is, up to a unit, a sum over $m$-tuples $(v_1,\dots,v_m)$
of total displacement zero of $m_{v_1}\cdots m_{v_m}$, where $d_v\equiv-m_v\lambda$
and $m_v=\psi$-exponent of $c(v)$. The involution
$(v_1,\dots,v_m)\mapsto(-v_1,\dots,-v_m)$ preserves the constraint, and
$c(-v)=-c(v)$ sends each $m_{v_i}$ to $-m_{v_i}$, multiplying the summand by
$(-1)^m=-1$. So the sum equals its own negative and vanishes in the residue
field, gaining one further power of $\lambda$.
\end{proof}

\begin{lemma}[the two closed forms]\label{lem:closed}
Put $Q=\sum_{v,w\neq0}d_vd_w\,\psi_t\bigl(-\omega(v,w)\bigr)$ and
$R=\tr(FDFD)/q$. Then, exactly,
\[
\tr H=qS,\quad \tr(D^2)=-2qS,\quad \tr(FD^2)=q\,(Q+2S),\quad
\tr(H^2)=qR+4qQ,
\]
and, modulo $\lambda^{4}$, $\;Q\equiv0\;$ and $\;R\equiv-2S$.
\end{lemma}

\begin{proof}
The exact identities follow as in Theorem~\ref{thm:main}, keeping track of
which products of $\rho$'s are central. For $Q$: expanding
$\psi_t(-\omega)\equiv1+\text{(linear in }\lambda)$ and using
$d_vd_w\equiv m_vm_w\lambda^2$, the constant term is
$\sum_{v,w}d_vd_w=S^2$, of valuation $\ge4$ since $\vlam(S)\ge2$; the surviving
term pairs the symmetric coefficient $d_vd_w$ against the antisymmetric
$\omega(v,w)$ and therefore cancels under $(v,w)\mapsto(w,v)$. For $R$:
reparametrizing so that the two free vectors are summed for fixed $(v,w)$, and
writing $s=-(v+w)$, the phase is $2\omega(x,v)+\omega(x,s)+\omega(v,s)$; summing
over the $q^2-2$ admissible $x$ and using
$\sum_{x\in\FF_q^2}\omega(x,v)=0$ gives
$2\omega(v,s)+0+(q^2-2)\omega(v,s)=q^{2}\omega(v,s)$, which vanishes in the
residue field. What survives is the count of admissible pairs, giving
$R\equiv-2S$.
\end{proof}

\begin{theorem}[the cancellation]\label{thm:cancel}
$2p_1\equiv p_2\pmod{\lambda^{\vlam(q)+4}}$; equivalently
$e_1+e_2\equiv0\pmod{\lambda^{\vlam(q)+4}}$.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:closed},
\[
2(q^2-1)\tr H-\tr(H^2)=q\bigl[\,2(q^2-1)S-R-4Q\,\bigr]
\equiv q\bigl[\,2(q^2-1)S+2S\,\bigr]=q\cdot 2q^2S \pmod{q\lambda^4},
\]
and $\vlam(2q^{2}S)=2\vlam(q)+\vlam(S)\ge4$. Dividing by $(q^2-1)^2$, a
$\lambda$-unit, gives $\vlam(2p_1-p_2)\ge\vlam(q)+4$. Finally
$\vlam(p_1^{2})\ge2(\vlam(q)+2)\ge\vlam(q)+4$, so
$e_2=(p_1^{2}-p_2)/2$ yields $e_1+e_2\equiv p_1-p_2/2\equiv0$.
\end{proof}

\begin{theorem}[the sharp law]\label{thm:sharp}
Suppose $\vlam(e_q)\ge\vlam(q)+4$. Then for every odd prime power $q=p^f$,
\[
\det B_t(c)\;\equiv\;\det F\pmod{\lambda^{\,\vlam(q)+4}} .
\]
The hypothesis is automatic whenever $f\ge2$, so for every odd prime power that
is not prime the conclusion is unconditional.
\end{theorem}

\begin{proof}[Proof that the hypothesis is automatic for $f\ge2$]
Every entry of $D$ is a $\ZZ[\zeta_p]$-combination of the $d_v$, each of
valuation $\ge1$, so $D=\lambda D'$ with $D'$ integral and
$\vlam(\det D)\ge q$. As $\vlam(\det F)=0$ we get $\vlam(e_q)\ge q$, and
$q\ge \vlam(q)+4=f(p-1)+4$ holds for every $f\ge2$ (at $q=9$: $9\ge8$; at
$q=27$: $27\ge10$; at $q=25$: $25\ge12$; the margin only grows). For $f=1$ the
inequality reads $q\ge q+3$ and fails, which is why the primes retain a
residual.
\end{proof}

\begin{proof}
Newton's identities and Lemmas~\ref{lem:power},~\ref{lem:parity} give
$\vlam(e_k)\ge\vlam(q)+4$ for $3\le k\le q-1$: the terms $e_{k-1}p_1$ and
$e_{k-2}p_2$ are products of two quantities of valuation $\ge\vlam(q)+2$ by
Theorem~\ref{thm:cong}, and $p_i$ for $i\ge3$ has valuation $\ge\vlam(q)+4$.
Theorem~\ref{thm:cancel} handles $k=1,2$ jointly and the hypothesis handles
$k=q$.
\end{proof}

So the only case still carrying a hypothesis is $q$ prime, where it asks for
$\vlam(\det D)\ge q+3$. Lemma~\ref{lem:top} gives $\vlam(e_q)\ge\vlam(q)+2$
and Newton's identities give $\vlam(\det D)\ge q+1$, both short by two. The
measured value is much larger:
\[
\min_c \vlam(\det D)=2q \qquad (6,\,10,\,14 \text{ at } q=3,5,7),
\]
with equality generically, so the Newton polygon of $D'=D/\lambda$ is the
straight line of slope $-1$ and every eigenvalue of $D$ has valuation exactly
$2$. Since $2q\ge q+3$ for every odd prime, the single statement

\begin{conjecture}\label{conj:top}
Every eigenvalue of $D$ has $\lambda$-valuation at least $2$; equivalently
$\vlam(\det D)\ge 2q$.
\end{conjecture}

\noindent closes the prime case and with it the whole law. In terms of the
elementary symmetric functions $e_k=e_k(D)$ the conjecture says exactly
$\vlam(e_k)\ge2k$ for all $k$, and all but the last case is a theorem.

\begin{theorem}[the eigenvalue bound, off the top]\label{thm:ek2k}
For every odd prime $q$ and every inverse-closed section,
$\vlam(e_k)\ge 2k$ for all $1\le k\le q-1$.
\end{theorem}

\begin{proof}
Induction on $k$. Since $p_1=\tr D=0$, Newton's identity reads
$k\,e_k=\sum_{i=2}^{k}(-1)^{i-1}e_{k-i}p_i$. By Lemma~\ref{lem:power}
$\vlam(p_i)\ge\vlam(q)+i$, and by induction $\vlam(e_{k-i})\ge2(k-i)$, so each
term has valuation at least
\[
2(k-i)+\vlam(q)+i \;=\; 2k-i+\vlam(q)\;\ge\;2k ,
\]
because $i\le k\le q-1=\vlam(q)$. For $k\le q-1$ the integer $k$ is a unit at
$\lambda$, so dividing by it is harmless.
\end{proof}

The induction fails at $k=q$ for one identifiable reason: there $i$ may equal
$q$, and the inequality $i\le\vlam(q)=q-1$ misses by exactly one. Accordingly
Newton yields only $\vlam(\det D)\ge q+1$ against the measured $2q$. So the
entire residual gap in the determinant law is the single coefficient
$e_q=\det D$ of one characteristic polynomial.

Two further pieces of evidence. First, $e_2=qS$ exactly, from
$e_2=(p_1^2-p_2)/2=-p_2/2$ and $p_2=-2qS$ (Lemma~\ref{lem:closed}); this is the
second exact closed form in the family after $\tr H=qS$. Second, $D'$ reduces
mod $\lambda$ to $\sum_a\kappa_aP_a$ with $\kappa_{-a}=-\kappa_a$ (so
$\kappa_0=0$ and $\sum_a\kappa_a=0$), which lies in the augmentation ideal of
$\FF_q[\ZZ/q]=\FF_q[s]/(s^{q})$ --- a nilpotent ideal. Hence every eigenvalue
of $D'$ is a nonunit, giving valuation $>0$; the conjecture upgrades this to
$\ge1$.

\begin{remark}
The exponent is attained: at $q=3$ every non-flat section gives valuation
exactly $6=\vlam(q)+4$, at $q=5$ the observed values are $\{8,10,12\}$ with
minimum $8$, at $q=7$ the minimum is $10$, and at $q=9$ --- where
$\vlam(9)=4$, not $q-1=8$ --- the minimum is $8$. All four agree with
$\vlam(q)+4$.
\end{remark}

\begin{remark}
The residue class is the flat determinant, not a numerical coincidence of small
$q$. At $q=3$ the two determinant values are $-16$ (flat orbit) and $11$, which
differ by $27$; since $(\lambda^{6})=(27)$ for $q=3$, the congruence
$\det B\equiv\det F\pmod{\lambda^{q+3}}$ there reads $\det B\equiv-16\pmod{27}$,
and $11\equiv-16$. An earlier guess that the residue is $q^2+2$ fails already
at $q=5$.
\end{remark}

\section{Prime powers}\label{sec:pf}

Let $q=p^f$ with $f>1$, $\psi_t(z)=\zeta_p^{\operatorname{Tr}_{\FF_q/\FF_p}(tz)}$,
and $\lambda=1-\zeta_p$. Lemma~\ref{lem:flat} survives verbatim: at $q=9$ one
finds $F^2+2F-80I=0$, spectrum $\{8^5,(-10)^4\}$, and
$\det F=8^5(-10)^4=327{,}680{,}000$, in agreement with the formula of
Lemma~\ref{lem:flat}; the trace laws hold as well.

Nothing else changes. Every ingredient of the argument --- the flat-block
Lemma~\ref{lem:flat}, the vanishing of $\tr\rho_t$ off the centre
(Lemma~\ref{lem:trace}), the pairing Lemma~\ref{lem:pair}, the parity
Lemma~\ref{lem:parity}, and the two closed forms of Lemma~\ref{lem:closed} ---
holds verbatim with $\psi_t$ in place of $\zeta^{t\,\cdot}$ and $\omega$ the
symplectic form over $\FF_q$. In particular the symplectic cancellations go
through because $\sum_{x\in\FF_q^2}\omega(x,v)=0$ for $v\neq0$ and because
$q^{2}\equiv0$ in the residue field, neither of which sees $f$.

The single quantity that depends on $f$ is the ramification,
\[
\vlam(q)=\vlam(p^{f})=f(p-1),
\]
since $(p)=(\lambda)^{p-1}$ in $\ZZ[\zeta_p]$. Theorems~\ref{thm:cong},
\ref{thm:cancel} and \ref{thm:sharp} were therefore already stated in their
correct generality: \emph{there is no separate prime-power theory}.

\begin{remark}[why this is one law and not two]
At prime $q$ we have $\vlam(q)=q-1$, so the sharp exponent
$\vlam(q)+4$ reads $q+3$; at $q=9$ we have $\vlam(9)=4$ and it reads $8$. As
functions of $q$ these look unrelated --- $8$ sits far below the value
$q+3=12$ that the prime formula would predict --- which is why we first
recorded the two cases as different phenomena, the prime case as a law and the
prime-power case as its failure. As functions of $\vlam(q)$ they are one
statement. The constant $4=2+2$ counts inverse closure twice, once at first
order and once in the $e_1/e_2$ cancellation, and is blind to the
factorization of $q$; only the ramification varies.
\end{remark}

The measured minima agree with $\vlam(q)+4$ at every prime power tested,
across three different values of $f$:
\begin{center}
\begin{tabular}{@{}lcccccccc@{}}
\toprule
$q$ & $3$ & $5$ & $7$ & $9$ & $25$ & $49$ & $27$ & $81$\\
$(p,f)$ & $(3,1)$ & $(5,1)$ & $(7,1)$ & $(3,2)$ & $(5,2)$ & $(7,2)$ & $(3,3)$ & $(3,4)$\\
\midrule
$\vlam(q)$ & $2$ & $4$ & $6$ & $4$ & $8$ & $12$ & $6$ & $8$\\
$\vlam(q)+4$ & $6$ & $8$ & $10$ & $8$ & $12$ & $16$ & $10$ & $12$\\
observed min & $6$ & $8$ & $10$ & $8$ & $12$ & $16$ & $10$ & $12$\\
\bottomrule
\end{tabular}
\end{center}
Eight prime powers across four values of $f$; the last five columns are
unconditional by Theorem~\ref{thm:sharp}. Note that $q$ and $\vlam(q)$ order
the columns differently --- $q=49$ has a larger exponent than $q=81$ --- which
is the clearest sign that the ramification, not $q$, is the governing
parameter. The determinants beyond $q=9$ were computed by fraction-free
(Bareiss) elimination over $\ZZ[\zeta_p]$, validated against cofactor
expansion at $q=3,5,7$; the flat determinants match Lemma~\ref{lem:flat}
exactly in every case, including the $35$- and $39$-digit values
$24^{13}26^{12}$ at $q=25$ and $-26^{14}28^{13}$ at $q=27$.

\subsection{The law does not survive over $\ZZ/p^{n}$}

It is natural to ask whether the field can be replaced by a finite chain ring.
It cannot, and the failure is instructive. Take $q=9$ but with the ring
$\ZZ/9$ in place of $\FF_9$: the group, the sections and the inverse-closure
condition are unchanged, and only the central character changes, from order
$3$ to order $9$. The construction remains valid --- $\rho$ is a homomorphism,
$\tr\rho$ vanishes off the centre, and the flat block still satisfies
$F^2+2F-80I=0$ with spectrum $\{8^5,(-10)^4\}$ and $\det F=327{,}680{,}000$,
numerically identical to the field case. But now $\lambda=1-\zeta_9$ and
$\vlam(9)=12$, so the law would predict exponent $16$.

The measured exponent is $12=\vlam(q)$: over $20$ sections the depths are
$\{12,16,18\}$. \emph{The entire $+4$ is lost.} The mechanism is exact, and it
is a single one. Newton's identity divides by $k$, which is harmless only when
$k$ is a $\lambda$-unit; and while $\vlam(3)=2$ in $\ZZ[\zeta_3]$, in
$\ZZ[\zeta_9]$ one has $\vlam(3)=6$. With block size $9$ the recursion passes
through $k=3,6,9$ in both cases, and only when the character has order $9$ is
the loss large enough to consume the whole $+4$.

It is tempting to blame the ring structure --- $\ZZ/9$ has zero divisors and
nonzero vectors that are not unimodular, so one might expect the symplectic
sums of Lemma~\ref{lem:closed} to degrade. \emph{They do not.} For any finite
Frobenius ring $R$ with generating character $\psi$ one has
$\sum_{a\in R}\psi(au)=0$ for every $u\neq0$ --- that is the defining property
--- so
\[
\sum_{x\in R^2}\psi\bigl(\omega(x,u)\bigr)
=\Bigl(\sum_{x_0}\psi(x_0u_1)\Bigr)\Bigl(\sum_{x_1}\psi(-u_0x_1)\Bigr)=0
\]
for every $u\neq0$, unimodular or not; we verified this exhaustively over
$\ZZ/9$. The character sums are perfectly healthy over the ring.

\begin{proposition}[the discriminator is the character order]\label{prop:scope}
What controls the $+4$ is the order of the generating character, not whether
the coefficient ring is a field.
\end{proposition}

The evidence is a third experiment, chosen to separate the two readings. Let
$R=\FF_3[x]/(x^2)$, a Frobenius ring which is \emph{not} a field --- it has
zero divisors and non-unimodular nonzero vectors --- but whose generating
character $\psi(c)=\zeta_3^{c_1}$ (the socle coordinate) has order $3$. Then
$|R|=9$ and $\vlam(9)=4$, exactly as over $\FF_9$. Running the same
computation: $\rho$ is a homomorphism, the flat block again satisfies
$F^2+2F-80I=0$ with $\det F=327{,}680{,}000$, and the measured depths are
$\{8,10,12\}$ with minimum $8=\vlam(q)+4$. \textbf{The $+4$ survives over a
non-field.}

So the trichotomy is: $\FF_9$ (field, character order $3$) gives $8$;
$\FF_3[x]/(x^2)$ (not a field, character order $3$) gives $8$; $\ZZ/9$ (not a
field, character order $9$) gives $12$. The field hypothesis plays no role;
Theorem~\ref{thm:sharp} should be read as a statement about coefficient rings
whose generating character has order $p$, and its proof goes through verbatim
in that generality.
